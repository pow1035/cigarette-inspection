cmake_minimum_required(VERSION 3.18)
project(YOLO26_ONNX_Inference CUDA CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)

# ONNX Runtime path
set(ONNXRUNTIME_ROOT "C:/Program Files/onnxruntime" CACHE PATH "ONNX Runtime安装路径")

# OpenCV path
set(OpenCV_DIR "C:/opencv/build" CACHE PATH "OpenCV安装路径")

find_package(OpenCV REQUIRED)
find_package(CUDA REQUIRED)

# Include directories
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${OpenCV_INCLUDE_DIRS}
    ${ONNXRUNTIME_ROOT}/include
    ${CUDA_INCLUDE_DIRS}
)

# Link directories
link_directories(
    ${ONNXRUNTIME_ROOT}/lib
    ${CUDA_TOOLKIT_ROOT_DIR}/lib/x64
)

# YOLO ONNX library
add_library(yolo_onnx STATIC
    yolo_onnx.cpp
    yolo_onnx.h
)

target_link_libraries(yolo_onnx
    ${OpenCV_LIBS}
    onnxruntime
    onnxruntime_providers_cuda
    onnxruntime_providers_shared
)

# Test executable
add_executable(test_yolo_onnx
    test_yolo_onnx.cpp
)

target_link_libraries(test_yolo_onnx
    yolo_onnx
    ${OpenCV_LIBS}
    onnxruntime
    onnxruntime_providers_cuda
    onnxruntime_providers_shared
)

# Visual Studio specific settings
if(MSVC)
    set_property(TARGET yolo_onnx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    set_property(TARGET test_yolo_onnx PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

    # Copy ONNX Runtime DLLs to output directory
    add_custom_command(TARGET test_yolo_onnx POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll"
        $<TARGET_FILE_DIR:test_yolo_onnx>
    )

    add_custom_command(TARGET test_yolo_onnx POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ONNXRUNTIME_ROOT}/lib/onnxruntime_providers_cuda.dll"
        $<TARGET_FILE_DIR:test_yolo_onnx>
    )

    add_custom_command(TARGET test_yolo_onnx POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ONNXRUNTIME_ROOT}/lib/onnxruntime_providers_shared.dll"
        $<TARGET_FILE_DIR:test_yolo_onnx>
    )
endif()

# Print configuration
message(STATUS "========================================")
message(STATUS "YOLO26 ONNX Runtime Configuration")
message(STATUS "========================================")
message(STATUS "ONNX Runtime: ${ONNXRUNTIME_ROOT}")
message(STATUS "OpenCV:       ${OpenCV_DIR}")
message(STATUS "CUDA:         ${CUDA_TOOLKIT_ROOT_DIR}")
message(STATUS "========================================")
